local function phase(t, a, b)
  local x = math.max(0, math.min(1, (t - a) / (b - a)))
  return x * x * (3 - 2 * x)
end

function pose(t, ctx)
  local n = #ctx.route
  local from = ctx.route[1]
  local home = ctx.route[n]
  local raise = phase(t, 0, 0.13) - phase(t, 0.93, 1)
  local reach = phase(t, 0.12, 0.23) - phase(t, 0.86, 0.94)
  local lower = phase(t, 0.23, 0.32) - phase(t, 0.79, 0.86)
  local insertion = phase(t, 0.32, 0.38) - phase(t, 0.73, 0.79)
  local lift = phase(t, 0.38, 0.47) - phase(t, 0.64, 0.73)
  local move = phase(t, 0.47, 0.64)
  local lx = from.x + (home.x - from.x) * move
  local lz = from.y + (home.y - from.y) * move
  local cargoY = 1.52 * lift
  local parkedX = -1.2
  local truckX = parkedX + (lx - parkedX) * raise
  local topY = 1.45 + (2.74 - 1.45) * raise
  local tipX = truckX + 0.4 + (lx - truckX - 0.4) * reach
  local forkY = (topY - 0.16) + ((cargoY + 0.09) - (topY - 0.16)) * lower
  local forkLen = 0.018 + (0.78 - 0.018) * insertion
  return {
    { id = 'truck', resource = 'Diorama/Lift', x = truckX, y = 0, z = lz },
    { id = 'tower', resource = 'Diorama/Lift', child = 'tower', x = truckX + 0.4, y = (1.24 + topY) / 2, z = lz, sy = topY - 1.24 },
    { id = 'boom', resource = 'Diorama/Lift', child = 'boom', x = (truckX + 0.4 + tipX) / 2, y = topY, z = lz, sx = math.max(0.001, tipX - truckX - 0.4) },
    { id = 'mast', resource = 'Diorama/Lift', child = 'mast', x = tipX, y = (topY + forkY) / 2, z = lz, sy = math.max(0.001, topY - forkY) },
    { id = 'fork', resource = 'Diorama/Lift', child = 'fork', x = tipX + forkLen / 2, y = forkY, z = lz, sx = forkLen },
    { id = ctx.boxId, resource = 'Diorama/Cargo', x = lx, y = cargoY, z = lz }
  }
end
